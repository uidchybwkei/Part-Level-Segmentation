# =============================================================================
# Semantic-SAM runtime requirements
# Tested on: NVIDIA A800-SXM4-80GB (sm_80), Python 3.11.9,
# PyTorch 2.8.0+cu129, CUDA 12.9, cuDNN 9.10.2
#
# IMPORTANT - install order:
#
#   1) Create env:
#        conda create -n semsam-py311-cu129 python=3.11.9 -y
#        conda activate semsam-py311-cu129
#
#   2) Install CUDA 12.9 toolkit for compiling CUDA extensions:
#        conda install -c nvidia/label/cuda-12.9.1 cuda-nvcc cuda-cudart-dev cuda-libraries-dev -y
#
#   3) Install PyTorch 2.8.0 + cu129:
#        pip install torch==2.8.0 torchvision==0.23.0 --index-url https://download.pytorch.org/whl/cu129
#
#   4) Install MaureenZOU's detectron2 fork:
#        export CUDA_HOME=$CONDA_PREFIX
#        export TORCH_CUDA_ARCH_LIST="8.0"
#        export FORCE_CUDA=1
#        pip install ninja
#        pip install --no-build-isolation -e ../detectron2-xyz
#
#   5) Install panopticapi:
#        pip install 'git+https://github.com/cocodataset/panopticapi.git'
#
#   6) Install the rest of the python deps:
#        pip install -r requirements.txt
#
#   7) Build the MultiScaleDeformableAttention CUDA op:
#        cd semantic_sam/body/encoder/ops && python setup.py build install && cd -
#
#   8) Run WebUI:
#        python fastapi_webui.py --ckpt swinl_only_sam_many2many.pth --model_type L --port 7860
#
# Notes:
#   * Pillow is pinned <10 because detectron2-xyz still uses the removed
#     `PIL.Image.LINEAR` constant. scikit-image is therefore pinned <0.22.
#   * numpy is pinned <2 because pycocotools / older numpy callers still use
#     deprecated np.float / np.int aliases.
# =============================================================================
